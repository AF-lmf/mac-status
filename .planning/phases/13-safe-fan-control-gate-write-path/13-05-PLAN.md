    private struct FanControlAffordanceView: View {
        let fanSnapshot: FanSnapshot
        let controlState: FanControlState
        @State private var pendingTarget: Double = 0  // initialized from live snapshot bounds midpoint

        // Computed bounds (unified: tightest across all fans per D-07)
        private var fanMin: Double {
            fanSnapshot.fans.compactMap(\.minRPM).max() ?? 0
        }
        private var fanMax: Double {
            fanSnapshot.fans.compactMap(\.maxRPM).min() ?? 0
        }
        private var boundsValid: Bool { fanMin < fanMax && fanMin > 0 }

        var body: some View {
            VStack(alignment: .leading, spacing: 8) {
                // Sub-header
                Text("手动控制")
                    .font(.caption)
                    .foregroundStyle(.secondary)

                // Mode row: label on left, mode text on label side (NOT value column — avoids jitter from variable string length)
                HStack {
                    Text("当前模式")
                        .font(.caption)
                        .foregroundStyle(.secondary)
                    Spacer()
                    Text(modeText)
                        .font(.caption)
                        .foregroundStyle(modeColor)
                        .lineLimit(1)
                }

                // Target row: label on left, fixed monospaced value column on right
                if case .manualVerified(let rpm) = controlState {
                    StableValueRow(label: "目标转速") {
                        StableValueText(
                            text: "\(Int(rpm)) RPM",
                            width: StableValueWidth.fanTargetRPM,
                            color: .primary,
                            font: .system(.body, design: .monospaced),
                            fontWeight: .medium
                        )
                    }
                }

                // Slider — only interactive when NOT pending; bounded to live [min, max]
                if boundsValid {
                    Slider(
                        value: $pendingTarget,
                        in: fanMin...fanMax,
                        step: 50
                    ) {
                        EmptyView()
                    } minimumValueLabel: {
                        EmptyView()
                    } maximumValueLabel: {
                        EmptyView()
                    } onEditingChanged: { editing in
                        // Commit on drag-end (editing transitions true → false) to avoid write storm
                        if !editing {
                            Task { @MainActor in
                                await FanControlManager.shared.setManualTarget(
                                    rpm: pendingTarget,
                                    snapshot: fanSnapshot
                                )
                            }
                        }
                    }
                    .disabled(controlState == .pending)
                    .accessibilityLabel("风扇目标转速，范围 \(Int(fanMin)) 到 \(Int(fanMax)) RPM")

                    // Bounds caption
                    StableCaptionText(text: "可调范围 \(Int(fanMin))–\(Int(fanMax)) RPM")
                }

                // Fail-closed inline message (per D-10: .secondary, NOT red, no modal)
                if case .failClosed(let message) = controlState {
                    StableCaptionText(text: message)
                }

                // Always-visible restore button (per D-11) — visible while manual mode is engaged
                if controlState == .manualVerified(targetRPM: pendingTarget)
                    || controlState == .pending
                    || (if case .failClosed = controlState { true } else { false }) {
                    // Simplified: show restore when NOT systemAuto and NOT unavailable
                    Button("恢复系统自动控制") {
                        Task { @MainActor in
                            await FanControlManager.shared.restoreAutoControl()
                        }
                    }
                    .foregroundStyle(Color.accentColor)
                    .font(.caption)
                }
            }
        }

        private var modeText: String {
            switch controlState {
            case .systemAuto: return "系统自动"
            case .pending: return "正在应用…"
            case .manualVerified: return "手动已生效"
            case .failClosed(let message): return message
            case .helperPendingApproval: return "等待授权…"
            case .unavailable: return "控制未启用"
            }
        }

        private var modeColor: Color {
            switch controlState {
            case .manualVerified: return .primary
            default: return .secondary  // all other states are calm secondary per UI-SPEC
            }
        }
    }

| T-13-05-02 | Information Disclosure | '手动已生效' shown before read-back | mitigate | FanControlAffordanceView reads controlState from DashboardState which is set only after FanControlManager.verifyManualMode returns true; IOKit success alone does not update the state |
| T-13-05-03 | Denial of Service | Continuous write storm from slider drag | mitigate | onEditingChanged: !editing guard (commit on drag-end only); slider disabled during .pending state; prevents flooding the privileged helper |
| T-13-05-04 | Denial of Service | View retains XPC connection | mitigate | FanControlAffordanceView has no NSXPCConnection; only FanControlManager.shared (per-call connection) is used; connection is created and invalidated per call in FanControlManager |
| T-13-05-05 | Tampering | SMC key strings in view layer | mitigate | DashboardView.swift and FanControlAffordanceView must not contain F0Md/F1Md/F0Tg/Ftst strings; verified via acceptance criteria grep |
