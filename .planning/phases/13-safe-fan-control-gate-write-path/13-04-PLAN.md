    if isSafeControlAvailable {
        // Toggle with side-effect: first enable shows dialog; subsequent enables go direct
        Toggle("风扇手动控制", isOn: Binding(
            get: { settings.manualFanControlEnabled },
            set: { newValue in
                if newValue && !settings.hasConfirmedFanControlRisk {
                    // First enable: show risk dialog before writing to settings
                    showingFanRiskDialog = true
                } else if newValue {
                    // Already confirmed: enable directly and register helper
                    settings.manualFanControlEnabled = true
                    Task { await MainActor.run { FanControlManager.shared.registerHelperIfNeeded() } }
                } else {
                    // Turning off: disable; FanControlManager observer will trigger restoreAutoControl
                    settings.manualFanControlEnabled = false
                }
            }
        ))
        .confirmationDialog(
            "开启手动风扇控制？",
            isPresented: $showingFanRiskDialog,
            titleVisibility: .visible
        ) {
            Button("开启手动控制", role: .destructive) {
                settings.hasConfirmedFanControlRisk = true
                settings.manualFanControlEnabled = true
                Task { await MainActor.run { FanControlManager.shared.registerHelperIfNeeded() } }
            }
            Button("取消", role: .cancel) {
                // Toggle reverts — we never wrote to settings.manualFanControlEnabled
                // so the Binding's get: returns false. No additional state change needed.
            }
        } message: {
            Text("手动控制会直接调整风扇转速。设置过低可能导致设备过热。\n\nMacStatus 会把目标转速限制在硬件安全范围内，不能低于系统默认下限，也不能让风扇停转。\n\n写入失败或退出应用时，MacStatus 会自动恢复系统自动控制。")
        }

        // Helper-approval state: show inline guide when SMAppService needs user action
        if FanControlManager.shared.currentState == .helperPendingApproval {
            VStack(alignment: .leading, spacing: 4) {
                Text("请在 系统设置 › 登录项 中允许 "MacStatus 风扇控制助手"")
                    .font(.caption2)
                    .foregroundStyle(.secondary)
                Button("打开系统设置") {
                    if let url = URL(string: "x-apple.systempreferences:com.apple.LoginItems-Settings.Extension") {
                        NSWorkspace.shared.open(url)
                    }
                }
                .buttonStyle(.link)
                .font(.caption2)
                .foregroundStyle(Color.accentColor)
            }
        }

        // Optional help caption (per UI-SPEC)
        Text("仅在受支持的 MacBook Pro 上可用，写入前会读回验证")
            .font(.caption2)
            .foregroundStyle(.secondary)
    }

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User interaction → SettingsView toggle | User-visible UI; opt-in is explicit; first-enable requires destructive dialog confirm |
| SettingsView → FanControlManager (privileged path trigger) | registerHelperIfNeeded() is the only escalation action triggered from UI |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-13-04-01 | Elevation of Privilege | 风扇手动控制 toggle triggering helper registration | mitigate | Toggle is hidden when safeControlAvailable == false (D-04); first enable requires explicit destructive confirm (D-02); registerHelperIfNeeded() only registers LaunchDaemon, not executes writes |
| T-13-04-02 | Information Disclosure | Dialog body misleading copy | mitigate | Copy locked in UI-SPEC; all three paragraphs (risk/clamp promise/restore promise) are required; developer must not soften them |
| T-13-04-03 | Tampering | hasConfirmedFanControlRisk flag bypass | mitigate | Flag is in UserDefaults (not secret), but bypassing it only skips the risk dialog — it does not bypass the capability gate or the helper registration flow; the dialog is an informed-consent UX, not a security boundary |
| T-13-04-04 | Denial of Service | 打开系统设置 URL opens wrong location | mitigate | URL "x-apple.systempreferences:com.apple.LoginItems-Settings.Extension" is the correct System Settings deep link for Login Items; user can dismiss if wrong |
| T-13-04-SC | Tampering | npm/pip/cargo installs | accept | No package manager installs |
</threat_model>
